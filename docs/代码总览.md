```js
import { mediaIndexEntry } from '../data/mediaIndex.generated.js';

// 前端不存放任何图片：所有图片都从云端链接取。
// 索引里的 url 就是云端那张图，hash 能在 url 里找到（见 docs/media-contract.md）。
const configuredBase = (import.meta.env.VITE_MEDIA_BASE_URL || '').replace(/\/$/, '');

/**
 * 取一张图的地址。
 * @param {string} activityId 活动编号，如 '01-1'
 * @param {string} relativePath 相对路径，从 `01_活动照片/` 起，如 '01_活动照片/10000_封面_海报_x.jpg'
 * @returns {string} 云端链接；索引里没有这张图时返回空字符串（调用方不要渲染）
 */
export function mediaUrl(activityId, relativePath) {
  const normalizedPath = String(relativePath || '').replace(/^\//, '');
  const entry = mediaIndexEntry(activityId, normalizedPath);
  if (entry) return entry.url;
  // 只有显式配置了本地媒体服务器时才回退到本地路径，正常情况永远走云端
  if (configuredBase) return `${configuredBase}/${activityId}/${normalizedPath}`;
  return '';
}

// 兼容旧调用：应用封面这类素材只有云端链接，没有本地文件。
export const cloudMediaUrl = mediaUrl;
```

```js
/**
 * 生成前端媒体索引。
 *
 * 设计原则：
 *   1. 候选集合 = 「网站压缩素材」现场扫描结果，也就是线上原本展示的那批图。不增不减。

 *   ../../素材库/<活动>/manifest.json           活动 id / 标题 / 日期 / 地点

function discoverFiles(root) {
  const result = [];
  if (!fs.existsSync(root)) return result;
  const stack = [root];
  while (stack.length) {
    const current = stack.pop();
    for (const entry of fs.readdirSync(current, { withFileTypes: true })) {
      const full = path.join(current, entry.name);
      if (entry.isDirectory()) stack.push(full);
      else result.push(path.relative(root, full).replaceAll(path.sep, '/'));
    }
  }
  return result;
}

function sha256File(filePath) {
  return crypto.createHash('sha256').update(fs.readFileSync(filePath)).digest('hex');
```

**作用**：只读文件头取像素尺寸，识别 1×1 占位白图
**行数**：55

```js
/**
 * 读图片像素尺寸，用来识别「1×1 白图」这类压缩坏掉的占位图。
 *
 * 为什么需要：压缩那一步会产出 631 字节的 1×1 白图（文件名正常、能打开），
 * 一旦传上云端就会在页面上显示成一个个白方块。上传前必须先把它们挡掉。
 * 只解析文件头，不依赖任何第三方库，也不解码整张图。
 */
import fs from 'node:fs';

/** @returns {{width:number,height:number}|null} 认不出来时返回 null（不拦） */
export function imageDimensions(filePath) {
  let fd;
  try {
    fd = fs.openSync(filePath, 'r');
    const head = Buffer.alloc(32);
    const read = fs.readSync(fd, head, 0, 32, 0);
    if (read < 16) return null;

    // PNG: 固定偏移 16 起是 IHDR 的宽高
    if (head.slice(0, 8).toString('hex') === '89504e470d0a1a0a') {
      return { width: head.readUInt32BE(16), height: head.readUInt32BE(20) };
    }

    // JPEG: 逐个 marker 找到 SOFn，宽高在 SOFn 段里
    if (head[0] === 0xff && head[1] === 0xd8) {
      const buffer = fs.readFileSync(filePath);
      let offset = 2;
      while (offset + 9 < buffer.length) {
        if (buffer[offset] !== 0xff) { offset += 1; continue; }
        const marker = buffer[offset + 1];
        const length = buffer.readUInt16BE(offset + 2);
        // SOF0..SOF3, SOF5..SOF7, SOF9..SOF11, SOF13..SOF15
        if (marker >= 0xc0 && marker <= 0xcf && ![0xc4, 0xc8, 0xcc].includes(marker)) {
          return { height: buffer.readUInt16BE(offset + 5), width: buffer.readUInt16BE(offset + 7) };
        }
        offset += 2 + length;
      }
      return null;
    }

    return null;
  } catch {
    return null;
  } finally {
    if (fd !== undefined) fs.closeSync(fd);
  }
}

/** 占位图判定：任一边 ≤ 2 像素 */
export function isPlaceholderImage(filePath) {
  const size = imageDimensions(filePath);
  if (!size) return false;
  return size.width <= 2 || size.height <= 2;
}
```

```js
  if (/案例/u.test(top)) return CATEGORY_CASE;
  if (/社交媒体/u.test(top)) return CATEGORY_PRESS;
  if (/活动/u.test(top)) return CATEGORY_ACTIVITY;
  return null;
}

// 从路径里找最近一个有意义的上层目录名，用作描述兜底
export function lastMeaningfulDir(relPath) {
  const parts = String(relPath || '').split('/').slice(0, -1);
  for (let index = parts.length - 1; index >= 0; index -= 1) {
    const part = parts[index].trim();
    if (!part || GENERIC_DIR.test(part)) continue;
    return part;
  }
  return '';
}

const startsWithAny = (segment, words) => words.find((word) => segment.startsWith(word)) || '';

/**
 * 解析一个文件名。
 * 文件名格式：`五位数字分数_内容分类_具体描述.扩展名`，段数不固定，
 * 分隔符可能是下划线、空格或短横线（例如 `10050 封面-活动照片_IMG_1.jpg`）。
 */
export function parseMediaName({ name, path = '', category = CATEGORY_ACTIVITY }) {
  const segments = splitNameSegments(name);
  const vocabulary = VOCABULARY[category] || VOCABULARY[CATEGORY_ACTIVITY];
  const notes = [];

  let rating = '00000';
  let rest = segments;
  if (RATING.test(segments[0] || '')) {
    rating = (segments[0] || '').slice(0, 5);
    // 评分后面可能直接贴着内容（10050封面活动.jpg），把剩下的部分留作一段
    const leftover = (segments[0] || '').slice(5).replace(LEADING_SEPARATORS, '');
    rest = leftover ? [leftover, ...segments.slice(1)] : segments.slice(1);
  } else {
    notes.push('缺评分');
  }

  let isCover = 0;
  let onHomePage = 0;
  let categoryText = '';
  const descriptionParts = [];

  for (const segment of rest) {
    const homepageMark = segment.startsWith(MARKER_HOMEPAGE);
    if (homepageMark) {
      onHomePage = 1;
      const leftover = segment.slice(MARKER_HOMEPAGE.length).replace(LEADING_SEPARATORS, '');
      if (leftover) descriptionParts.push(leftover);
      continue;
    }
    // 「封面」只在活动里代表活动封面；案例里的「封面」是案例自己的封面，算细分类
    if (segment.startsWith(MARKER_COVER) && category === CATEGORY_ACTIVITY) {
      isCover = 1;
      const leftover = segment.slice(MARKER_COVER.length).replace(LEADING_SEPARATORS, '');
      if (leftover) descriptionParts.push(leftover);
      continue;
    }
    const word = startsWithAny(segment, vocabulary);
    if (!categoryText && word) {
      categoryText = word;
      const leftover = segment.slice(word.length).replace(LEADING_SEPARATORS, '');
      if (leftover) descriptionParts.push(leftover);
      continue;
    }
    descriptionParts.push(segment);
  }

  let unclassifiedSegments = [];
  if (!categoryText) {
    categoryText = UNCLASSIFIED;
    notes.push('未分类');
    unclassifiedSegments = descriptionParts.slice();
  }

  let description = descriptionParts.join('_').trim();
  if (!description) {
    description = lastMeaningfulDir(path);
    if (description) notes.push('描述兜底为目录名');
    else {
      description = String(name || '').replace(/\.[^./]+$/u, '');
      notes.push('描述兜底为文件名');
    }
  }

  return { rating, isCover, onHomePage, categoryText, description, notes, unclassifiedSegments };
}
```

```jsx
import { useEffect, useMemo, useRef, useState } from 'react';

const ratioOf = (item, dimensions) => dimensions[item.id]?.src === item.image ? dimensions[item.id].ratio : 1.5;

function buildRows(items, dimensions, width, targetHeight, gap) {
  if (!width) return [];
  const rows = [];
  let current = [];
  let ratioSum = 0;
  const flush = (fillWidth) => {
    if (!current.length) return;
    const available = width - gap * (current.length - 1);
    const fittedHeight = available / ratioSum;
    // A short final row must stay at the normal card height and align left.
    // Only a row with enough content is justified to the full container width.
    const height = fillWidth ? fittedHeight : Math.min(targetHeight, fittedHeight);
    rows.push({ items: current, height });
    current = [];
    ratioSum = 0;
  };

  items.forEach((item) => {
    const ratio = ratioOf(item, dimensions);
    current.push(item);
    ratioSum += ratio;
    const projected = ratioSum * targetHeight + gap * (current.length - 1);
    if (current.length >= 2 && projected >= width) flush(true);
  });
  flush(false);
  return rows;
}

export default function JustifiedGallery({ items, label, targetHeight = 240, onOpen, renderCaption, renderOverlay, fitToWidth = true, multiRow = false, firstRowOnly = false, onVisibleCountChange }) {
  const rootRef = useRef(null);
  const [width, setWidth] = useState(0);
  const [dimensions, setDimensions] = useState({});

  useEffect(() => {
    const root = rootRef.current;
    if (!root) return undefined;
    const update = () => setWidth(root.clientWidth);
    update();
    const observer = new ResizeObserver(update);
    observer.observe(root);
    return () => observer.disconnect();
  }, []);

  useEffect(() => {
    let cancelled = false;
    items.forEach((item) => {
      if (dimensions[item.id]?.src === item.image) return;
      const image = new Image();
      image.onload = () => {
        if (cancelled || !image.naturalWidth || !image.naturalHeight) return;
        setDimensions((current) => ({
          ...current,
          [item.id]: { src: item.image, ratio: image.naturalWidth / image.naturalHeight },
        }));
      };
      image.src = item.image;
    });
    return () => { cancelled = true; };
  }, [dimensions, items]);

  const row = useMemo(() => {
    const gap = 12;
    const ratioSum = items.reduce((sum, item) => sum + ratioOf(item, dimensions), 0);
    const naturalHeight = (width - gap * Math.max(items.length - 1, 0)) / ratioSum;
    // Never enlarge a sparse strip just to consume the entire row. This keeps
    // one case (or a few cases) at the same visual size as a populated row.
    const height = fitToWidth ? Math.max(1, Math.min(targetHeight, naturalHeight)) : targetHeight;
    return { height, items };
  }, [dimensions, fitToWidth, items, targetHeight, width]);

  const rows = useMemo(() => {
    if (!multiRow && !firstRowOnly) return [row];
    const calculated = buildRows(items, dimensions, width, targetHeight, 12);
    return firstRowOnly ? calculated.slice(0, 1) : calculated;
  }, [dimensions, items, multiRow, firstRowOnly, row, targetHeight, width]);
  const visibleCount = rows.reduce((count, current) => count + current.items.length, 0);
  useEffect(() => { onVisibleCountChange?.(visibleCount); }, [onVisibleCountChange, visibleCount]);
  const captionHeight = renderCaption ? 48 : 0;

  const registerImage = (item, event) => {
    const image = event.currentTarget;
    if (!image.naturalWidth || !image.naturalHeight) return;
    setDimensions((current) => ({
      ...current,
      [item.id]: { src: item.image, ratio: image.naturalWidth / image.naturalHeight },
    }));
  };

  return (
    <div className="justified-gallery" ref={rootRef} aria-label={label}>
      {rows.map((currentRow, rowIndex) => <div
        className={`justified-row ${multiRow ? 'justified-row-multiple' : 'justified-row-single'}`}
        style={{ height: currentRow.height + captionHeight }}
        key={rowIndex}
      >
        {currentRow.items.map((item) => (
          <div
            className="justified-card"
            style={{
              width: `${ratioOf(item, dimensions) * currentRow.height}px`,
              height: currentRow.height + captionHeight,
            }}
            key={item.id}
          >
            <button className="justified-item" style={{ height: currentRow.height }} type="button" onClick={() => onOpen(item)}>
              <img src={item.image} alt={item.alt} onLoad={(event) => registerImage(item, event)} loading="lazy" />
              {renderOverlay ? renderOverlay(item) : null}
            </button>
            {renderCaption ? renderCaption(item) : null}
          </div>
        ))}
        </div>)}
    </div>
  );
}
```

```jsx
      if (event.key === 'Escape') onClose();
      if (event.key === 'ArrowLeft' && index > 0) onChange(index - 1);
      if (event.key === 'ArrowRight' && index < items.length - 1) onChange(index + 1);
    };
    document.body.classList.add('modal-open');
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.body.classList.remove('modal-open');
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [index, item, items.length, onChange, onClose]);

  if (!item) return null;
  return (
    <div className="lightbox" role="dialog" aria-modal="true" aria-label={`图片预览 ${index + 1} / ${items.length}`} onMouseDown={(event) => event.target === event.currentTarget && onClose()}>

      <IconButton className="lightbox-prev" label="上一张图片" disabled={index === 0} onClick={() => onChange(index - 1)}><ChevronLeft /></IconButton>
      <figure className="lightbox-figure">
        <img src={item.image} alt={item.alt} style={{ objectPosition: item.position }} />
```

```jsx
import { ChevronLeft, ChevronRight } from 'lucide-react';
import { useCallback, useEffect, useRef, useState } from 'react';
import IconButton from './IconButton.jsx';

export default function Carousel({ items, label, renderItem }) {
  const railRef = useRef(null);
  const [paused, setPaused] = useState(false);
  const [position, setPosition] = useState({ start: true, end: false, index: 0 });

  const updatePosition = useCallback(() => {
    const rail = railRef.current;
    if (!rail) return;
    const max = rail.scrollWidth - rail.clientWidth;
    const firstCard = rail.firstElementChild;
    const step = firstCard ? firstCard.getBoundingClientRect().width + 20 : rail.clientWidth;
    setPosition({
      start: rail.scrollLeft <= 2,
      end: rail.scrollLeft >= max - 2,
      index: Math.min(items.length - 1, Math.round(rail.scrollLeft / step)),
    });
  }, [items.length]);

  const move = useCallback((direction) => {
    const rail = railRef.current;
    const card = rail?.firstElementChild;
    if (!rail || !card) return;
    const amount = card.getBoundingClientRect().width + 20;
    rail.scrollBy({ left: direction * amount, behavior: 'smooth' });
  }, []);

  useEffect(() => {
    const rail = railRef.current;
    if (!rail) return undefined;
    updatePosition();
    rail.addEventListener('scroll', updatePosition, { passive: true });
    window.addEventListener('resize', updatePosition);
    return () => {
      rail.removeEventListener('scroll', updatePosition);
      window.removeEventListener('resize', updatePosition);
    };
  }, [updatePosition]);

  useEffect(() => {
    if (paused || items.length < 2 || window.matchMedia('(prefers-reduced-motion: reduce)').matches) return undefined;
    const timer = window.setInterval(() => {
      const rail = railRef.current;
      if (!rail) return;
      if (rail.scrollLeft >= rail.scrollWidth - rail.clientWidth - 2) rail.scrollTo({ left: 0, behavior: 'smooth' });
      else move(1);
    }, 5600);
    return () => window.clearInterval(timer);
  }, [items.length, move, paused]);

  return (
    <div className="carousel" onMouseEnter={() => setPaused(true)} onMouseLeave={() => setPaused(false)}>
      <div className="carousel-rail" ref={railRef} aria-label={label} tabIndex="0">
        {items.map((item, index) => renderItem(item, index))}
      </div>
      <IconButton className="carousel-arrow carousel-arrow-left" label="查看上一项" disabled={position.start} onClick={() => move(-1)}>
        <ChevronLeft aria-hidden="true" />
      </IconButton>
      <IconButton className="carousel-arrow carousel-arrow-right" label="查看下一项" disabled={position.end} onClick={() => move(1)}>
        <ChevronRight aria-hidden="true" />
      </IconButton>
      <div className="carousel-dots" aria-hidden="true">
        {items.map((item, index) => <span key={item.id} className={index === position.index ? 'is-active' : ''} />)}
      </div>
    </div>
  );
}
```

```js
import { useCallback, useEffect, useState } from 'react';

function readRoute() {
  const routePath = window.location.hash.startsWith('#/')
    ? window.location.hash.slice(1)
    : window.location.pathname;
  const match = routePath.match(/^\/events\/([^/]+)(?:\/(gallery|production-gallery))?\/?$/);
  if (!match) return { page: 'archive', eventId: null };
  const page = match[2] === 'gallery' ? 'gallery' : match[2] === 'production-gallery' ? 'production-gallery' : 'detail';
  return { page, eventId: decodeURIComponent(match[1]) };
}

export default function useRoute() {
  const [route, setRoute] = useState(readRoute);

  useEffect(() => {
    const handlePopState = () => setRoute(readRoute());
    window.addEventListener('popstate', handlePopState);
    window.addEventListener('hashchange', handlePopState);
    return () => {
      window.removeEventListener('popstate', handlePopState);
      window.removeEventListener('hashchange', handlePopState);
    };
  }, []);

  const navigate = useCallback((path, options = {}) => {
    const needsHashRoute = window.location.protocol === 'file:'
      || window.location.pathname.includes('/api/files/html-preview/')
      || window.location.pathname.endsWith('/index.html');
    const target = needsHashRoute ? `${window.location.pathname}${window.location.search}#${path}` : path;
    window.history[options.replace ? 'replaceState' : 'pushState']({}, '', target);
    setRoute(readRoute());
    window.scrollTo({ top: 0, behavior: options.instant ? 'auto' : 'smooth' });
  }, []);

  return { route, navigate };
}
```

## scripts/activity-selection.test.mjs

**行数**：97

```js
import assert from 'node:assert/strict';
import { parseMediaName, categoryOfPath, isImageName } from '../src/lib/mediaNaming.js';
import { selectActivityPhotos } from '../src/lib/activitySelection.js';

/* ---------------------------------------------------------------- 名字解析 */

// 评分：必须是开头连续五位数字
assert.equal(parseMediaName({ name: '10000_封面_海报_x.jpg' }).rating, '10000');
assert.equal(parseMediaName({ name: '首页 (2).jpg' }).rating, '00000');
assert.equal(parseMediaName({ name: '09880_大合照_x.jpg' }).rating, '09880');

// 「封面」标记：下划线、空格、短横线三种写法都要认
for (const name of ['10050 封面 合照.jpg', '10050封面活动.jpg', '10050_封面-合照.jpg', '10050_封面 大合照_x.jpg']) {
  assert.equal(parseMediaName({ name, category: '活动' }).isCover, 1, `应为封面: ${name}`);
}

// 「首页」标记
for (const name of ['09900_首页_互动.jpg', '09880 首页 活动照片_IMG_2195.jpg', '首页.jpg']) {
  assert.equal(parseMediaName({ name, category: '活动' }).onHomePage, 1, `应进首页: ${name}`);
}

// 案例里的「封面」是案例自己的封面，不算活动封面
const caseCover = parseMediaName({ name: '08750_封面_PawMatch.jpg', category: '案例' });
assert.equal(caseCover.isCover, 0);
assert.equal(caseCover.categoryText, '封面');

// 细分类必须落在词表里，落不进去记「其他」
assert.equal(parseMediaName({ name: '09580_人物_参与者微笑.jpg', category: '活动' }).categoryText, '人物');
assert.equal(parseMediaName({ name: '09350_活动照片_微信图片_x.jpg', category: '活动' }).categoryText, '其他');
assert.equal(parseMediaName({ name: '08760_作品界面_PawMatch_05.jpg', category: '案例' }).categoryText, '作品界面');

// 描述：拼接剩余段；空了退到目录名，再退到文件名
assert.equal(parseMediaName({ name: '07486_演讲_活动现场记录_IMG_4580.jpg', category: '活动' }).description, '活动现场记录_IMG_4580');
assert.equal(parseMediaName({ name: '主图.jpg', path: '02_案例/08640_AI宠写真/截图/主图.jpg', category: '案例' }).description, '08640_AI宠写真');

// 分类由第一层目录决定，「待确认_案例素材」要落到案例
assert.equal(categoryOfPath('01_活动照片/a.jpg'), '活动');
assert.equal(categoryOfPath('02_案例/x/a.jpg'), '案例');
assert.equal(categoryOfPath('02_案例_旧镜像_2026/x.jpg'), '案例');
assert.equal(categoryOfPath('03_社交媒体/a.jpg'), '社交媒体');
assert.equal(categoryOfPath('待确认_案例素材/a.jpg'), '案例');
assert.equal(categoryOfPath('随便/a.jpg'), null);

// 图片判定：mp4 不算图片
assert.equal(isImageName('a.JPG'), true);
assert.equal(isImageName('a.mp4'), false);

/* ------------------------------------------------------------------ 选图 */

const activity = (name, extra = {}) => ({
  activityId: '01-1',
  path: `01_活动照片/${name}`,
  name,
  category: '活动',
  categoryText: '其他',
  rating: parseMediaName({ name, category: '活动' }).rating,
  description: '',
  isCover: 0,
  onHomePage: 0,
  url: 'https://example.test/x.jpg',
  hash: 'x',
  originBytes: 1,
  cloudBytes: 1,
  ...extra,
});

const marked = (name) => {
  const parsed = parseMediaName({ name, category: '活动' });
  return activity(name, { isCover: parsed.isCover, onHomePage: parsed.onHomePage });
};

const lawyers = ['首页.jpg', '首页 (2).jpg', '09900_首页_互动.jpg', '09800_现场.jpg'].map(marked);
assert.equal(selectActivityPhotos(lawyers).highlights.length, 3);

const photos = ['10000_首页_海报.jpg', '09900_封面_合照.jpg', '09800 首页人物.jpg', '09700 普通.jpg'].map(marked);
const selected = selectActivityPhotos(photos);
assert.equal(selected.cover.name, photos[1].name);
assert.deepEqual(selected.highlights.map((a) => a.name), [photos[1].name, photos[0].name, photos[2].name]);
assert.equal(selected.sorted[0].name, photos[0].name);

// 一张活动同时有 10050 和 10000 两个封面时，取评分高的那个（与改造前一致）
const twoCovers = ['10050 封面 活动照片_a.jpg', '10000_封面_海报_b.jpg'].map(marked);
assert.equal(selectActivityPhotos(twoCovers).cover.name, '10050 封面 活动照片_a.jpg');

const agent = ['10050_封面_海报_短图.jpg', '10000__海报_长图.jpg', '09300_首页_互动.jpg'].map(marked);
assert.equal(selectActivityPhotos(agent).highlights.length, 2);
assert.equal(selectActivityPhotos(agent).cover.name, agent[0].name);

assert.equal(selectActivityPhotos([marked('09900_首页_互动.jpg'), marked('09800_现场.jpg')]).cover.name, '09900_首页_互动.jpg');
// 视频不参与选图
assert.equal(selectActivityPhotos([activity('10050_封面_视频.mp4', { category: '活动', isCover: 1 })]).highlights.length, 0);
assert.equal(selectActivityPhotos([marked('09900_首页_IMG_1.jpg'), marked('09800_首页_IMG_2.jpg')]).highlights.length, 2);
// 案例图不参与活动选图
assert.equal(selectActivityPhotos([{ ...activity('封面_图.jpg'), category: '案例', path: '02_案例/封面_图.jpg' }]).sorted.length, 0);

console.log('媒体命名 + 选图回归检查通过。');
```
